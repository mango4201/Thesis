% ── FRAME: uncertainty models ── [1.5 min]
% Takeaway: three adversaries of increasing subtlety.
% Source: Ch2 def:discrete-uncertainty, def:interval-uncertainty,
%         def:budgeted-uncertainty
\begin{frame}{Three Adversaries}
  \begin{block}{Definition (uncertainty sets)}
    \begin{itemize}
      \item Discrete: $\Scenarios = \{\cs{1}, \ldots, \cs{K}\}$, one full scenario is chosen
      \item Interval: $\Scenarios = \prod_{e \in E} [\ell_e, u_e]$, every edge varies independently
      \item Budgeted: intervals, but at most $\Gamma$ edges may deviate upwards
    \end{itemize}
  \end{block}
  % TODO: click-through TikZ, one micro-graph edge under each model
  % Transition: "Fix the adversary; what should 'best tree' mean?"
\end{frame}

% ── FRAME: robust objectives ── [1.5 min]
% Takeaway: two notions of protection, genuinely different optima.
% Source: Ch2 eq:minmax-objective, eq:regret-def,
%         eq:minmax-regret-objective, tab:micro-graph-costs
\begin{frame}{Two Robust Objectives}
  \begin{block}{Min-max}
    \[ \min_{T \in \cT}\; \max_{c \in \Scenarios}\; c(T) \]
  \end{block}
  \begin{block}{Min-max regret}
    \[ \min_{T \in \cT}\; \max_{c \in \Scenarios}\; \bigl( c(T) - \MSTcost{c} \bigr) \]
  \end{block}
  \begin{itemize}
    \item<2-> Absolute worst case vs.\ worst case measured against hindsight
    \item<3-> On the micro-graph the two objectives pick \alert{different} trees
    % TODO: concrete trees per objective ONLY after re-verification
    %       against tab:micro-graph-costs + fresh Python enumeration
  \end{itemize}
  % Transition: "We take the two objectives in turn, min-max first,
  % from the easiest adversary to the hardest."
\end{frame}

\section{Min-Max Spanning Trees}

% ── FRAME: interval extremal lemma, proved on-slide ── [3 min]
% Takeaway: against the interval adversary, one MST call suffices.
% Source: Ch3 lem:interval-extremal-cost, cor:mm-interval-polynomial;
%         Goerigk Thm 4.8
\begin{frame}{Intervals: the Adversary Has No Real Choice}
  \begin{block}{Lemma}
    For every spanning tree $T$ it holds that $\wc{T} = \sum_{e \in E(T)} u_e$.
  \end{block}
  \begin{itemize}
    \item<2-> Edges vary independently, the cost is increasing in every $c_e$
    \item<3-> So the worst case sets each tree edge to its upper bound
    \item<4-> Hence: min-max tree $=$ MST under $u$ \textcolor{RWTHGreen}{$\Rightarrow$ polynomial}
  \end{itemize}
  \medskip
  \onslide<5->{{\scriptsize Source: thesis Lemma 3.1; cf.\ Goerigk, Theorem 4.8.}}
  % TODO: click-through on the micro-graph (tree edges pushed to u_e,
  %       non-tree edges greyed out)
  % Transition: "One scenario per edge was harmless. What if the
  % adversary holds complete scenarios instead? Two already suffice."
\end{frame}

% ── FRAME: K=2 hardness statement ── [1 min]
% Takeaway: two scenarios flip the problem to NP-hard.
% Source: Ch3 thm:mm-k2-hard; Goerigk Thm 8.4
\begin{frame}{Discrete Scenarios: Two Are Already Hard}
  \begin{block}{Theorem}
    The min-max spanning tree problem with $K = 2$ discrete scenarios is $\mathsf{NP}$-hard.
  \end{block}
  \begin{itemize}
    \item<2-> Reduction from the \emph{Partition} problem
  \end{itemize}
  \medskip
  \onslide<3->{{\scriptsize Source: thesis Theorem 3.2; cf.\ Goerigk, Theorem 8.4.}}
\end{frame}

% ── FRAME: budgeted intuition ── [1 min; D4: may move to appendix]
% Takeaway: a budget on deviations restores tractability via pricing.
% Source: Ch3 thm:mm-budgeted-poly; Goerigk Thm 4.21, Lemma 4.20
\begin{frame}{Why the Budget Helps: Pricing the Adversary}
  \begin{itemize}
    \item Guess a price $\pi$ per unit of deviation budget
    \item<2-> Solve one \alert{ordinary} MST under costs $\bar{c}_e + [\hat{c}_e - \pi]^{+}$, pay $\Gamma\pi$
    \item<3-> Only $m + 1$ candidate prices matter \textcolor{RWTHGreen}{$\Rightarrow$ polynomial}
  \end{itemize}
  \medskip
  \onslide<4->{{\scriptsize Source: thesis \S3.4 (in progress); cf.\ Goerigk, Theorem 4.21.}}
